# `paddock doctor` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A read-only diagnosis of the request path to a site — daemon, ports, resolver, DNS, Caddy, PHP, folder, certificate, HTTP and newest log error — in the CLI, the menu-bar app and a read-only MCP tool, with the one command that fixes each failure.

**Architecture:** One pure core value type, `Doctor`, takes the current `EngineStatus` and four injected probes and returns a `DoctorReport` of outcome rows. The CLI, `DoctorModel` (app) and the MCP tool each fetch a status, call `Doctor.report`, and render it; nothing is written, fixed or restarted, and no daemon IPC changes.

**Tech Stack:** Swift 6 compiler in Swift 5 language mode, SwiftUI/AppKit (macOS 13+), XCTest, Swift ArgumentParser, `URLSession` for one HTTP probe.

**Spec:** `docs/superpowers/specs/2026-11-25-doctor-design.md` (corrected before this plan: a `linked` check would have failed every parked site, so it is gone; the site report carries host/PHP/URL for the CLI heading; ports failing also skip Caddy and PHP-FPM).

## Global Constraints

- Doctor is read-only: it never runs a fix, writes a file, restarts a service, installs trust, or sends a changing request. Every fix is a command printed for the owner.
- No daemon IPC changes: `Request`, `Response` and `EngineStatus` stay as they are.
- Reuse existing wording and plumbing: `Messages`, `PortHolder.stopCommand`, `LogCatalog.siteLogs`, `MCPLogReader.tail`, `LogParser`, `PaddockBuild.number`, `Theme`.
- macOS 13+ (the package's floor); no new dependencies; XCTAssert-style tests.
- Do not bump `VERSION` or `PaddockBuild.number`; do not publish or merge.
- Every commit message is one plain sentence, no conventional-commit prefix, and hooks are never skipped.
- All existing tests stay green.

```swift
import Foundation

/// One row of a doctor report: what was looked at, what was found, and the command that fixes it.
public struct DoctorCheck: Codable, Equatable, Sendable, Identifiable {
    public enum Outcome: String, Codable, Sendable {
        case passed, failed, skipped, info
    }

    public let id: String
    /// What was looked at, without a final period: `PHP 8.5 is installed`.
    public let title: String
    public let outcome: Outcome
    /// One sentence: what was found.
    public let detail: String
    /// A command for the owner to run; nil when nothing helps or the check passed.
    public let fix: String?

    public init(id: String, title: String, outcome: Outcome, detail: String, fix: String? = nil) {
        self.id = id
        self.title = title
        self.outcome = outcome
        self.detail = detail
        self.fix = fix
    }

    static func passed(_ id: String, _ title: String, _ detail: String) -> DoctorCheck {
        DoctorCheck(id: id, title: title, outcome: .passed, detail: detail)
    }

    static func failed(_ id: String, _ title: String, _ detail: String, fix: String?) -> DoctorCheck {
        DoctorCheck(id: id, title: title, outcome: .failed, detail: detail, fix: fix)
    }

    static func skipped(_ id: String, _ title: String, _ detail: String) -> DoctorCheck {
        DoctorCheck(id: id, title: title, outcome: .skipped, detail: detail)
    }

    static func info(_ id: String, _ title: String, _ detail: String) -> DoctorCheck {
        DoctorCheck(id: id, title: title, outcome: .info, detail: detail)
    }
}

/// What one run of the doctor found, Paddock-wide and per site.
public struct DoctorReport: Codable, Equatable, Sendable {
    public struct SiteReport: Codable, Equatable, Sendable {
        public let site: String
        public let host: String
        public let php: PHPVersion
        public let url: String
        public let checks: [DoctorCheck]

        public init(site: String, host: String, php: PHPVersion, url: String, checks: [DoctorCheck]) {
            self.site = site
            self.host = host
            self.php = php
            self.url = url
            self.checks = checks
        }
    }

    public let global: [DoctorCheck]
    public let sites: [SiteReport]

    public init(global: [DoctorCheck], sites: [SiteReport]) {
        self.global = global
        self.sites = sites
    }

    public var failures: [DoctorCheck] {
        global.filter { $0.outcome == .failed }
            + sites.flatMap { $0.checks.filter { $0.outcome == .failed } }
    }

    public var isHealthy: Bool { failures.isEmpty }

    /// The report for a daemon whose status could not be fetched: one failed check, no site names.
    public static func unreachable(_ detail: String) -> DoctorReport {
        DoctorReport(global: [.failed("daemon", "The daemon answers", detail, fix: "paddock start")], sites: [])
    }
}

/// One HTTP answer, as the probe saw it.
public struct DoctorProbe: Equatable, Sendable {
    public let statusCode: Int
    public let milliseconds: Int

    public init(statusCode: Int, milliseconds: Int) {
        self.statusCode = statusCode
        self.milliseconds = milliseconds
    }
}

/// What a site's own log says, for the row that never fails.
public enum DoctorLogFinding: Equatable, Sendable {
    /// There is no `storage/logs/*.log`.
    case none
    case noErrors(file: String)
    case error(file: String, line: String)
    case unreadable(file: String)
}
```

```swift
import Foundation

/// The CLI's exact lines for a doctor report, so the format is tested without running the program.
public enum DoctorText {
    /// With `site`, that site's global and own checks under one heading; without it, Paddock's checks
    /// once and then each site, a healthy site as one line.
    public static func lines(for report: DoctorReport, site: String? = nil) -> [String] {
        var lines: [String] = []
        if site != nil {
            if let one = report.sites.first {
                lines.append(heading(one))
            } else {
                lines.append(site!)
            }
            lines.append("")
            for check in report.global + (report.sites.first?.checks ?? []) {
                append(check, to: &lines)
            }
        } else {
            lines.append("Paddock")
            lines.append("")
            for check in report.global { append(check, to: &lines) }
            for one in report.sites {
                lines.append("")
                if one.checks.contains(where: { $0.outcome == .failed }) {
                    lines.append(heading(one))
                    for check in one.checks { append(check, to: &lines) }
                } else {
                    lines.append("\(one.host): OK.")
                }
            }
        }
        lines.append("")
        lines.append(summary(report))
        return lines
    }

    private static func heading(_ site: DoctorReport.SiteReport) -> String {
        "\(site.host) — PHP \(site.php), \(site.url)"
    }

    private static func append(_ check: DoctorCheck, to lines: inout [String]) {
        switch check.outcome {
        case .passed:
            lines.append("  ✓ \(sentence(check.title))")
        case .failed:
            lines.append("  ✗ \(sentence(check.title))")
            lines.append("      \(sentence(check.detail))")
            if let fix = check.fix { lines.append("      Fix: \(fix)") }
        case .skipped:
            lines.append("  - \(sentence(check.title))")
            lines.append("      \(sentence(check.detail))")
        case .info:
            lines.append("  i \(sentence(check.title))")
            lines.append("      \(sentence(check.detail))")
        }
    }

    private static func sentence(_ text: String) -> String {
        guard let last = text.last, !".!?".contains(last) else { return text }
        return text + "."
    }

    private static func summary(_ report: DoctorReport) -> String {
        switch report.failures.count {
        case 0: return "Everything checks out."
        case 1: return "1 check needs attention."
        case let count: return "\(count) checks need attention."
        }
    }
}
```

```swift
/// Diagnoses the path from a request to a site. Pure: every outside fact comes from a probe.
public struct Doctor: Sendable {
    public typealias Probe = @Sendable (URL) async throws -> DoctorProbe
    public typealias LogFindingProbe = @Sendable (Site) -> DoctorLogFinding

    let build: Int
    let resolveHost: @Sendable (String) -> [String]
    let readableFolder: @Sendable (String) -> Bool
    let probeURL: Probe
    let newestLogError: LogFindingProbe
    let siteURL: @Sendable (Site) -> URL

    public init(build: Int,
                resolve: @escaping @Sendable (String) -> [String],
                readable: @escaping @Sendable (String) -> Bool,
                probe: @escaping Probe,
                logFinding: @escaping LogFindingProbe,
                url: @escaping @Sendable (Site) -> URL) {
        self.build = build
        self.resolveHost = resolve
        self.readableFolder = readable
        self.probeURL = probe
        self.newestLogError = logFinding
        self.siteURL = url
    }

    /// Every check, in order. `named` limits the site rows (the caller has already resolved it).
    public func report(for status: EngineStatus, site named: String? = nil) async -> DoctorReport {
        if let stopped = stoppedReport(status) { return stopped }

        let sites = status.sites.filter { named == nil || $0.name == named }
        var global: [DoctorCheck] = []
        var globalFailures: [String: String] = [:]

        func add(_ check: DoctorCheck) {
            global.append(check)
            if check.outcome == .failed { globalFailures[check.id] = check.title }
        }

        func skipped(_ id: String, _ title: String, because causes: [String]) -> DoctorCheck? {
            guard let cause = causes.first(where: { globalFailures[$0] != nil }) else { return nil }
            return .skipped(id, title, "Skipped: \(globalFailures[cause]!) failed.")
        }

        add(.passed("daemon", "The daemon answers (build \(status.version))",
                    "The daemon answers (build \(status.version))."))

        if status.blockedBy.isEmpty {
            add(.passed("ports", "Ports 80 and 443 are free", "Ports 80 and 443 are free."))
        } else {
            add(.failed("ports", "Ports 80 and 443 are free", Messages.cantStart(status.blockedBy),
                        fix: status.blockedBy.compactMap(\.stopCommand).first ?? "paddock start"))
        }

        if status.portHelperInstalled {
            add(.passed("port-helper", "The port helper manages 80 and 443",
                        "The port helper manages 80 and 443."))
        } else {
            add(.failed("port-helper", "The port helper manages 80 and 443",
                        "The port helper isn't installed.", fix: "paddock start"))
        }

        if status.resolverInstalled {
            add(.passed("resolver", "The .test resolver entry is installed",
                        "The .test resolver entry is installed."))
        } else {
            add(.failed("resolver", "The .test resolver entry is installed",
                        "The .test resolver entry isn't installed.", fix: "paddock start"))
        }

        let dnsSite = named == nil ? status.sites.first : sites.first
        if let dnsSite {
            let title = "DNS answers \(dnsSite.host)"
            let addresses = resolveHost(dnsSite.host)
            if addresses.contains("127.0.0.1") {
                add(.passed("dns", title, "\(dnsSite.host) resolves to 127.0.0.1."))
            } else {
                let found = addresses.isEmpty
                    ? "\(dnsSite.host) does not resolve."
                    : "\(dnsSite.host) resolves to \(addresses.joined(separator: ", "))."
                add(.failed("dns", title, found, fix: "paddock restart"))
            }
        } else {
            add(.skipped("dns", "DNS answers a site's host", "No sites to resolve."))
        }

        let caddyTitle = "Caddy is running"
        if let skip = skipped("caddy", caddyTitle, because: ["ports"]) {
            add(skip)
        } else if status.processes["caddy"]?.hasPrefix("running") == true {
            add(.passed("caddy", caddyTitle, "Caddy is running."))
        } else {
            add(.failed("caddy", caddyTitle, "Caddy isn't running.", fix: "paddock start"))
        }

        var siteReports: [DoctorReport.SiteReport] = []
        for site in sites {
            siteReports.append(await siteReport(for: site, in: status, globalFailures: globalFailures))
        }
        return DoctorReport(global: global, sites: siteReports)
    }

    /// A daemon that can't be diagnosed further: unreachable, stopped, or another build. One row only.
    private func stoppedReport(_ status: EngineStatus) -> DoctorReport? {
        let title = "The daemon answers (build \(status.version))"
        if status.version != build {
            return DoctorReport(global: [.failed("daemon", title,
                "The daemon is build \(status.version); this is build \(build).", fix: "paddock restart")],
                sites: [])
        }
        if !status.running && status.blockedBy.isEmpty {
            return DoctorReport(global: [.failed("daemon", title, "Paddock is stopped.",
                                                fix: "paddock start")], sites: [])
        }
        return nil
    }

    private func siteReport(for site: Site, in status: EngineStatus,
                            globalFailures: [String: String]) async -> DoctorReport.SiteReport {
        var checks: [DoctorCheck] = []
        var failures: [String: String] = [:]

        func add(_ check: DoctorCheck) {
            checks.append(check)
            if check.outcome == .failed { failures[check.id] = check.title }
        }

        func skipped(_ id: String, _ title: String, because causes: [String]) -> DoctorCheck? {
            var all = globalFailures
            for (key, value) in failures { all[key] = value }
            guard let cause = causes.first(where: { all[$0] != nil }) else { return nil }
            return .skipped(id, title, "Skipped: \(all[cause]!) failed.")
        }

        let phpTitle = "PHP \(site.php) is installed"
        if status.installedPHP.contains(site.php) {
            add(.passed("php", phpTitle, "PHP \(site.php) is installed."))
        } else {
            add(.failed("php", phpTitle, "PHP \(site.php) isn't installed.",
                        fix: "paddock use \(site.php)"))
        }

        let fpmTitle = "PHP-FPM \(site.php) is running"
        if let skip = skipped("php-fpm", fpmTitle, because: ["ports", "php"]) {
            add(skip)
        } else if status.processes["php-fpm-\(site.php)"]?.hasPrefix("running") == true {
            add(.passed("php-fpm", fpmTitle, "PHP-FPM \(site.php) is running."))
        } else {
            add(.failed("php-fpm", fpmTitle, "PHP-FPM \(site.php) isn't running.",
                        fix: "paddock restart"))
        }

        let folderTitle = "The site folder is readable"
        if readableFolder(site.path) {
            add(.passed("folder", folderTitle, "\(site.path) can be read."))
        } else {
            add(.failed("folder", folderTitle, "\(site.path) can't be read.", fix: nil))
        }

        let trustTitle = "macOS trusts Paddock's certificate"
        if !site.secured {
            add(.skipped("trust", trustTitle, "\(site.name) isn't secured; nothing to trust."))
        } else if !status.trustEnabled {
            add(.failed("trust", trustTitle, "Paddock can't install certificate trust in this build.",
                        fix: nil))
        } else if status.needsTrust {
            add(.failed("trust", trustTitle, "macOS doesn't trust Paddock's certificate.",
                        fix: "paddock trust"))
        } else {
            add(.passed("trust", trustTitle, "macOS trusts Paddock's certificate."))
        }

        let httpTitle = "\(site.url) answers"
        if let skip = skipped("http", httpTitle,
                              because: ["ports", "port-helper", "dns", "caddy", "php", "php-fpm"]) {
            add(skip)
        } else {
            add(await httpCheck(site, title: httpTitle))
        }

        add(logCheck(for: site))
        return DoctorReport.SiteReport(site: site.name, host: site.host, php: site.php, url: site.url,
                                       checks: checks)
    }

    private func httpCheck(_ site: Site, title: String) async -> DoctorCheck {
        let scheme = site.secured ? "HTTPS" : "HTTP"
        do {
            let probe = try await probeURL(siteURL(site))
            let found = "\(scheme) \(probe.statusCode) in \(probe.milliseconds) ms."
            if (500..<600).contains(probe.statusCode) {
                return .failed("http", title,
                               found + " Some request reached the site and failed; see the log row.",
                               fix: nil)
            }
            return .passed("http", title, found)
        } catch let error as URLError {
            let detail: String
            let fix: String?
            switch error.code {
            case .serverCertificateUntrusted, .serverCertificateHasUnknownRoot,
                 .serverCertificateHasBadDate, .serverCertificateNotYetValid,
                 .serverCertificate, .clientCertificateRejected, .clientCertificateRequired,
                 .secureConnectionFailed:
                detail = "TLS failed: \(error.localizedDescription)"
                fix = "paddock trust"
            case .cannotFindHost:
                detail = "\(site.host) did not resolve for the request."
                fix = "paddock restart"
            case .cannotConnectToHost:
                detail = "Nothing answered on port \(Self.port(of: siteURL(site)))."
                fix = "paddock start"
            case .timedOut:
                detail = "No answer within 5 seconds."
                fix = "paddock restart"
            default:
                detail = error.localizedDescription
                fix = nil
            }
            return .failed("http", title, detail, fix: fix)
        } catch {
            return .failed("http", title, error.localizedDescription, fix: nil)
        }
    }

    private func logCheck(for site: Site) -> DoctorCheck {
        let title = "Newest error in the site's log"
        switch newestLogError(site) {
        case .none:
            return .skipped("log", title, "No logs in storage/logs.")
        case .noErrors(let file):
            return .info("log", title, "No errors in \(file).")
        case .error(let file, let line):
            let trimmed = line.trimmingCharacters(in: .whitespaces)
            return .info("log", title, "Newest error in \(file): \(String(trimmed.prefix(160)))")
        case .unreadable(let file):
            return .skipped("log", title, "\(file) couldn't be read.")
        }
    }

    private static func port(of url: URL) -> Int {
        if let port = url.port { return port }
        return url.scheme == "https" ? 443 : 80
    }
}
```

```swift
import Darwin
import Foundation

extension Doctor {
    /// The numeric addresses `host` resolves to (getaddrinfo).
    public static func probeAddresses(_ host: String) -> [String] {
        var hints = addrinfo(ai_flags: AI_ADDRCONFIG, ai_family: AF_UNSPEC, ai_socktype: SOCK_STREAM,
                             ai_protocol: 0, ai_addrlen: 0, ai_canonname: nil, ai_addr: nil, ai_next: nil)
        var found: UnsafeMutablePointer<addrinfo>?
        guard getaddrinfo(host, nil, &hints, &found) == 0, let first = found else { return [] }
        defer { freeaddrinfo(first) }
        var addresses: [String] = []
        var node: UnsafeMutablePointer<addrinfo>? = first
        while let current = node {
            var text = [CChar](repeating: 0, count: Int(NI_MAXHOST))
            if getnameinfo(current.pointee.ai_addr, current.pointee.ai_addrlen, &text,
                           socklen_t(text.count), nil, 0, NI_NUMERICHOST) == 0 {
                let address = String(cString: text)
                if !addresses.contains(address) { addresses.append(address) }
            }
            node = current.pointee.ai_next
        }
        return addresses
    }

    /// A folder that exists, is a folder, and can be read.
    public static func folderIsReadable(_ path: String) -> Bool {
        var isDirectory: ObjCBool = false
        return FileManager.default.fileExists(atPath: path, isDirectory: &isDirectory)
            && isDirectory.boolValue
            && FileManager.default.isReadableFile(atPath: path)
    }

    /// The site's address, on Paddock's configured ports when they are not 80/443.
    public static func siteAddress(_ site: Site, settings: Settings = .current) -> URL {
        var components = URLComponents()
        components.scheme = site.secured ? "https" : "http"
        components.host = site.host
        let port = site.secured ? settings.httpsPort : settings.httpPort
        if port != (site.secured ? 443 : 80) { components.port = port }
        return components.url ?? URL(string: site.url)!
    }

    /// One GET with a 5-second timeout and default TLS trust: what a browser would get.
    public static func probeHTTP(_ url: URL) async throws -> DoctorProbe {
        let configuration = URLSessionConfiguration.ephemeral
        configuration.timeoutIntervalForRequest = 5
        configuration.requestCachePolicy = .reloadIgnoringLocalCacheData
        let session = URLSession(configuration: configuration)
        defer { session.invalidateAndCancel() }
        let clock = ContinuousClock()
        let start = clock.now
        let (_, response) = try await session.data(for: URLRequest(url: url))
        let elapsed = start.duration(to: clock.now)
        guard let http = response as? HTTPURLResponse else { throw URLError(.badServerResponse) }
        let milliseconds = Int(elapsed.components.seconds * 1000)
            + Int(elapsed.components.attoseconds / 1_000_000_000_000_000)
        return DoctorProbe(statusCode: http.statusCode, milliseconds: milliseconds)
    }

    /// The newest error in the site's own log, read the way the Logs window and `logs_read` read it.
    public static func logFindingFor(_ site: Site) -> DoctorLogFinding {
        let root = URL(fileURLWithPath: site.path, isDirectory: true)
        guard let source = LogCatalog.siteLogs(in: root).first else { return .none }
        let file = source.url.lastPathComponent
        guard let tail = MCPLogReader.tail(of: source.url, count: 4000, containing: nil, inside: root) else {
            return .unreadable(file: file)
        }
        var entries: [LogEntry] = []
        var parser = LogParser()
        parser.append(tail.lines, to: &entries)
        guard let error = entries.last(where: { $0.isProblem }) else { return .noErrors(file: file) }
        return .error(file: file, line: error.text)
    }
}
```

```swift
import ArgumentParser
import Foundation
import PaddockCore

struct DoctorCommand: ParsableCommand {
    static let configuration = CommandConfiguration(
        commandName: "doctor",
        abstract: "Check every step from a request to a site, and say what is wrong.")

    @Argument(help: "A site's name or host; every site when left out.") var site: String?

    func run() throws {
        let status: EngineStatus
        do {
            switch try CLI.client.send(.status, timeout: 10) {
            case .ok(let answer), .removed(_, let answer), .databaseCreated(_, _, let answer):
                status = answer
            case .error(_, let message):
                throw CLI.failure(message)
            case .catalog, .envLines:
                throw CLI.unexpectedAnswer()
            }
        } catch let code as ExitCode {
            throw code
        } catch ClientError.daemonNotRunning {
            try finish(.unreachable(Messages.daemonNotRunning))
        } catch ClientError.timedOut {
            try finish(.unreachable(Messages.didNotAnswer))
        } catch ClientError.developmentBuild {
            try finish(.unreachable(Messages.developmentBuild))
        } catch {
            try finish(.unreachable(Messages.unexpectedAnswer(error)))
        }

        let named = try site.map { try CLI.site(named: $0, in: status).name }
        let report = makeReport(for: status, site: named)
        print(DoctorText.lines(for: report, site: named).joined(separator: "\n"))
        if !report.isHealthy { throw ExitCode.failure }
    }

    /// Prints an unreachable-daemon report and ends with the failure exit code.
    private func finish(_ report: DoctorReport) throws -> Never {
        print(DoctorText.lines(for: report).joined(separator: "\n"))
        throw ExitCode.failure
    }

    /// `Doctor.report` is async; ArgumentParser's command is not, so this bridges it like `paddock mcp` does.
    private func makeReport(for status: EngineStatus, site: String?) -> DoctorReport {
        final class Box: @unchecked Sendable { var report: DoctorReport? }
        let box = Box()
        let finished = DispatchSemaphore(value: 0)
        Task.detached {
            box.report = await Doctor().report(for: status, site: site)
            finished.signal()
        }
        finished.wait()
        return box.report ?? .unreachable(Messages.didNotAnswer)
    }
}
```

```swift
import Combine
import Foundation

/// The Doctor window's state: the sites to choose from, the site chosen, and the last report.
@MainActor
public final class DoctorModel: ObservableObject {
    public typealias StatusFetcher = @Sendable () async throws -> EngineStatus
    public typealias Diagnoser = @Sendable (EngineStatus, String?) async -> DoctorReport

    @Published public private(set) var sites: [Site] = []
    /// Nil means every site.
    @Published public var target: String?
    @Published public private(set) var report: DoctorReport?
    @Published public private(set) var isChecking = false
    /// The status failure's sentence, also shown as the daemon row; nil after a successful check.
    @Published public private(set) var error: String?

    private let fetchStatus: StatusFetcher
    private let diagnose: Diagnoser

    public init(fetchStatus: @escaping StatusFetcher,
                diagnose: @escaping Diagnoser = { status, site in
                    await Doctor().report(for: status, site: site)
                }) {
        self.fetchStatus = fetchStatus
        self.diagnose = diagnose
    }

    /// A fresh status, then a fresh report. A failed fetch becomes the daemon row, never a blank error screen.
    public func check() async {
        guard !isChecking else { return }
        isChecking = true
        defer { isChecking = false }
        do {
            let status = try await fetchStatus()
            sites = status.sites
            if let target, !status.sites.contains(where: { $0.name == target }) { self.target = nil }
            report = await diagnose(status, target)
            error = nil
        } catch {
            let sentence: String
            let clientError = error as? ClientError
            if clientError == .daemonNotRunning {
                sentence = Messages.daemonNotRunning
            } else if clientError == .timedOut {
                sentence = Messages.didNotAnswer
            } else if clientError == .developmentBuild {
                sentence = Messages.developmentBuild
            } else {
                sentence = Messages.unexpectedAnswer(error)
            }
            error = sentence
            report = .unreachable(sentence)
        }
    }
}
```

```swift
import Foundation

extension MCPTools {
    /// The `doctor` tool: resolve the site the arguments name, diagnose, and answer as JSON.
    func diagnose(_ arguments: JSONValue) async throws -> MCPToolResult {
        let name = try arguments.optionalText("site"), path = try arguments.optionalText("path")
        let status = try currentStatus()
        let named: String?
        switch (name, path) {
        case (let name?, nil):
            named = try site(named: name, in: status).name
        case (nil, let path?):
            named = try site(containing: path, in: status).name
        case (nil, nil):
            named = nil
        default:
            throw MCPFailure("Give \"site\" or \"path\", not both.")
        }
        let report = await doctor.report(for: status, site: named)
        return .ok(json(report, site: named))
    }

    static func json(_ check: DoctorCheck) -> JSONValue {
        .object([
            "id": .string(check.id),
            "title": .string(check.title),
            "outcome": .string(check.outcome.rawValue),
            "detail": .string(check.detail),
            "fix": check.fix.map(JSONValue.string) ?? .null,
        ])
    }

    static func json(_ report: DoctorReport, site: String?) -> JSONValue {
        .object([
            "site": site.map(JSONValue.string) ?? .null,
            "global": .array(report.global.map(json)),
            "sites": .array(report.sites.map { one in
                .object([
                    "site": .string(one.site),
                    "checks": .array(one.checks.map(json)),
                ])
            }),
        ])
    }
}
```
